class PuzzleValidator: """복수 정답을 허용하는 퍼즐 검증기"""

  def validate_puzzle_order(
      self,
      user_order: List[int],
      correct_orders: List[List[int]],  # 복수 정답
      blocks: List[dict],
  ) -> dict:
      """
      검증 로직:
      1. 정확한 순서 일치 체크
      2. 함수 단위 교환 허용 체크
      3. 독립 블록 순서 교환 허용 체크
      """

      # 1. 정확한 일치
      for correct in correct_orders:
          if user_order == correct:
              return {"is_correct": True, "match_type": "exact"}

      # 2. 함수 단위 교환 체크
      user_code = self._assemble_code(user_order, blocks)
      for correct in correct_orders:
          correct_code = self._assemble_code(correct, blocks)
          if self._are_functionally_equivalent(user_code, correct_code):
              return {"is_correct": True, "match_type": "functional"}

      # 3. 독립 블록 체크 (같은 들여쓰기 레벨의 연속 블록)
      if self._check_independent_block_swap(user_order, correct_orders[0], blocks):
          return {"is_correct": True, "match_type": "independent_swap"}

      return {"is_correct": False, "expected": correct_orders[0]}

  def _are_functionally_equivalent(self, code1: str, code2: str) -> bool:
      """
      두 코드가 기능적으로 동일한지 체크
      - 함수 정의 순서만 다른 경우
      - import 순서만 다른 경우
      """
      # AST 비교 또는 정규화 후 비교
      normalized1 = self._normalize_code(code1)
      normalized2 = self._normalize_code(code2)
      return normalized1 == normalized2

  def _normalize_code(self, code: str) -> str:
      """코드 정규화 (함수 순서 정렬, 공백 제거)"""
      import ast
      try:
          tree = ast.parse(code)
          # 함수 정의를 이름순으로 정렬
          functions = sorted(
              [n for n in tree.body if isinstance(n, ast.FunctionDef)],
              key=lambda f: f.name
          )
          # ... 정규화 로직
          return ast.dump(tree)
      except:
          return code.strip()

async def evaluate_personalization_node(state: PersonalizedRAGState) -> dict: """검색 결과 개인화 품질 평가 (Agentic)""" results = state["search_results"] user_profile = state["user_profile"] skill_by_topic = user_profile.get("skill_by_topic", {})

  issues = []

  # 1. 난이도 적합성 체크
  requested_diff = state["difficulty"]
  if requested_diff:
      # 사용자 실력 대비 난이도 체크
      topic_skills = [skill_by_topic.get(t, 0.5) for t in state["topics"]]
      avg_skill = sum(topic_skills) / len(topic_skills) if topic_skills else 0.5

      diff_map = {"easy": 0.3, "medium": 0.5, "hard": 0.7, "very_hard": 0.9}
      requested_level = diff_map.get(requested_diff, 0.5)

      if avg_skill < 0.3 and requested_level > 0.5:
          issues.append("too_hard_for_skill_level")
      elif avg_skill > 0.8 and requested_level < 0.3:
          issues.append("too_easy_for_skill_level")

  # 2. 약점 주제 커버리지 체크
  weak_topics = state["weak_topics"]
  if weak_topics:
      result_topics = set()
      for r in results:
          result_topics.update(r.get("tags", []))

      weak_coverage = len(set(weak_topics) & result_topics) / len(weak_topics)
      if weak_coverage < 0.3:
          issues.append("weak_topics_not_covered")

  if issues and state.get("refinement_count", 0) < 2:
      return {
          "search_quality": "refine_needed",
          "refinement_reason": issues[0],
      }

  return {"search_quality": "good"}

async def refine_search_node(state: PersonalizedRAGState) -> dict: """검색 조건 개선""" reason = state["refinement_reason"]

  if reason == "too_hard_for_skill_level":
      # 난이도 낮춤
      new_difficulty = {"hard": "medium", "very_hard": "hard", "medium": "easy"}
      return {
          "difficulty": new_difficulty.get(state["difficulty"], state["difficulty"]),
          "refinement_count": state.get("refinement_count", 0) + 1,
      }

  elif reason == "weak_topics_not_covered":
      # 약점 주제 명시적 추가
      return {
          "topics": state["topics"] + state["weak_topics"][:1],
          "refinement_count": state.get("refinement_count", 0) + 1,
      }

  return {"refinement_count": state.get("refinement_count", 0) + 1}

  workflow.add_node("load_profile", load_user_profile_node)
  workflow.add_node("search", personalized_search_node)
  workflow.add_node("evaluate", evaluate_personalization_node)
  workflow.add_node("refine", refine_search_node)
  workflow.add_node("respond", respond_node)

  workflow.set_entry_point("load_profile")
  workflow.add_edge("load_profile", "search")
  workflow.add_edge("search", "evaluate")

  # Agentic 루프
  workflow.add_conditional_edges(
      "evaluate",
      lambda s: "refine" if s["search_quality"] == "refine_needed" else "respond",
      {"refine": "refine", "respond": "respond"}
  )
  workflow.add_edge("refine", "search")  # 재검색
  workflow.add_edge("respond", END)

  return workflow.compile()
